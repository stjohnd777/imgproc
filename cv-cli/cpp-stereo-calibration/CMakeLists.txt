cmake_minimum_required(VERSION 3.16)
project(stereo_calibration_cli LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(OpenCV REQUIRED COMPONENTS core imgproc imgcodecs calib3d)
find_package(nlohmann_json REQUIRED)
foreach(tool stereo_calibrate stereo_rectify)
    add_executable(${tool}_cli ${tool}_cli.cpp)
    target_link_libraries(${tool}_cli PRIVATE ${OpenCV_LIBS} nlohmann_json::nlohmann_json)
    target_include_directories(${tool}_cli PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../common)
endforeach()
include(CTest)
if(BUILD_TESTING)
    add_executable(stereo_calibration_fixture stereo_calibration_fixture.cpp)
    target_link_libraries(stereo_calibration_fixture PRIVATE ${OpenCV_LIBS})
endif()
