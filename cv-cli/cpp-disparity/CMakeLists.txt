cmake_minimum_required(VERSION 3.16)
project(disparity_cli LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(OpenCV REQUIRED COMPONENTS core imgproc imgcodecs calib3d)
add_executable(disparity_cli disparity_cli.cpp)
target_link_libraries(disparity_cli PRIVATE ${OpenCV_LIBS})
target_include_directories(disparity_cli PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../common /opt/homebrew/include)
